30	 * 递归遍历素材。
31	 *
32	 * `roots` 里**既可以是目录也可以是视频文件本身**：界面上用户是在系统对话框里
33	 * 直接挑 `.mp4` 的（多选），挑出来的就是一串文件路径；命令行则习惯给一个目录。
34	 * 两种都得认，否则"挑了 5 个文件，扫出 0 个"——还不报错。
35	 *
36	 * 显式点名的文件仍然要过扩展名白名单：挑中一个 `.txt` 不该被塞进素材库，
37	 * 后面 ffprobe 只会失败。
38	 *
39	 * 三条刻意的取舍：
40	 * - **跳过符号链接**：目录软链会成环，走进去就出不来了
41	 * - **跳过点开头的条目**：`.DS_Store`、`.Trash`、`@eaDir` 这类不是素材
42	 * - **单个目录读失败不中断整轮**：一个权限不足的子目录不该让 3000 个文件白扫
43	 *
44	 * 返回值按路径排序，让两次扫描的处理顺序稳定——续跑时更好对照进度。
45	 */
46	export async function scanMediaDirectory(
47	    roots: readonly string[],
48	    options: ScanOptions = {}
49	): Promise<ScannedFile[]> {
50	    const allowed = new Set(
51	        (options.extensions ?? DEFAULT_VIDEO_EXTENSIONS).map((e) =>
52	            e.toLowerCase()
53	        )
54	    );
55	    const found: ScannedFile[] = [];
56	    const seen = new Set<string>();
57	
58	    const addFile = async (full: string): Promise<void> => {
59	        if (!allowed.has(path.extname(full).toLowerCase())) return;
60	        if (seen.has(full)) return;
61	
62	        let info;
63	        try {
64	            info = await stat(full);
65	        } catch {
66	            return;
67	        }
68	
69	        const file: ScannedFile = {
70	            path: full,
71	            sizeBytes: info.size,
72	            // mtimeMs 是浮点。列是 integer，不取整会出现"存进去 1.7e12，读出来差 0.5"的假变更
73	            mtimeMs: Math.round(info.mtimeMs)
74	        };
75	        seen.add(full);
76	        found.push(file);
77	        options.onFound?.(file);
78	    };
79	
80	    const walk = async (dir: string): Promise<void> => {
81	        options.signal?.throwIfAborted();
82	
83	        let entries;
84	        try {
85	            entries = await opendir(dir);
86	        } catch {
87	            // 读不了就跳过这棵子树，不影响其余目录
88	            return;
89	        }
90	
91	        for await (const entry of entries) {
92	            options.signal?.throwIfAborted();

96	            if (entry.isSymbolicLink()) continue;
97	
98	            if (entry.isDirectory()) {
99	                await walk(full);
100	                continue;
101	            }
102	            if (!entry.isFile()) continue;
103	            await addFile(full);
104	        }
105	    };
106	
107	    for (const root of roots) {
108	        options.signal?.throwIfAborted();
109	        const full = path.resolve(root);
110	
111	        let info;
112	        try {
113	            info = await stat(full);
114	        } catch {
115	            // 目录被删了、盘没挂上：跳过这一项，别让其余的白扫
116	            continue;
117	        }
118	
119	        if (info.isDirectory()) {
120	            await walk(full);
121	        } else if (info.isFile()) {
122	            await addFile(full);
123	        }
124	    }
125

118	/** 片段 id 由 `路径 + 区间` 决定，同一个文件重扫得到同一个 id——续跑时不会产生孤儿行 */
119	export function segmentIdFor(
120	    filePath: string,
121	    startMs: number,
122	    durationMs: number
123	): string {
124	    const digest = createHash('sha1')
125	        .update(`${filePath}|${startMs}|${durationMs}`)
126	        .digest('hex');
127	    return `seg_${digest.slice(0, 16)}`;
128	}

214	     * 登记一个扫描到的文件，返回它是新增、变更还是没动。
215	     *
216	     * 判定为「变更」时把探测与抽帧状态**双双重置为 pending**，并清掉旧的关键帧与片段。
217	     * 不清的话，一个被重新导出过的文件会留着旧时长——下游据此裁出来的 clip 会超出源片长度。
218	     */
219	    upsertScannedFile(file: ScannedFile): UpsertResult {
220	        const key = contentKeyOf(file);
221	        const now = Date.now();
222	        const existing = this.db.get(
223	            'select id, content_key from media_files where path = ?',
224	            [file.path]
225	        ) as Row | null;
226	
227	        if (!existing) {
228	            const result = this.db.run(
229	                `insert into media_files (path, size_bytes, mtime_ms, content_key, updated_at)
230	                 values (?, ?, ?, ?, ?)`,
231	                [file.path, file.sizeBytes, file.mtimeMs, key, now]
232	            );
233	            return { id: Number(result.lastInsertRowid), status: 'added' };
234	        }
235	
236	        const id = Number(existing['id']);
237	        if (existing['content_key'] === key) {
238	            return { id, status: 'unchanged' };
239	        }
240	
241	        this.db.run(
242	            `update media_files
243	                set size_bytes = ?, mtime_ms = ?, content_key = ?,
244	                    probe_state = 'pending', probe_error = null,
245	                    duration_ms = null, width = null, height = null,

259	     *
260	     * **只处理本次扫描根目录下的路径。** 否则"这次只扫了目录 A"会把目录 B 的索引全删了。
261	     * 删的是索引行与关键帧记录，不动用户的任何素材文件。
262	     */

264	        roots: readonly string[],
265	        seenPaths: ReadonlySet<string>
266	    ): number {
267	        const resolved = roots.map((r) => path.resolve(r));
268	        const rows = this.db.all('select id, path from media_files') as Row[];
269	        let removed = 0;
270	        for (const row of rows) {
271	            const filePath = asString(row['path']);
272	            if (filePath === null || seenPaths.has(filePath)) continue;
273	            const underRoot = resolved.some(
274	                (root) =>
275	                    filePath === root || filePath.startsWith(root + path.sep)
276	            );
277	            if (!underRoot) continue;
278	            this.db.run('delete from media_files where id = ?', [
279	                Number(row['id'])
280	            ]);
281	            removed += 1;
282	        }
283	        return removed;

483	    /**
484	     * plan.md §8 的检索：**先硬过滤，再 BM25 排序**。
485	     *
486	     * 查询串为空（或切不出任何 token）时退化为纯过滤 + 按时长降序——
487	     * 空串交给 MATCH 是语法错误，必须走另一条分支。
488	     */
489	    search(query: string, filters: SearchFilters = {}): SearchHit[] {
490	        const limit = filters.limit ?? 20;
491	        const where: string[] = [];
492	        const params: (string | number)[] = [];
493	
494	        const match = toMatchExpression(query);
495	        if (match !== null) {
496	            where.push('asset_fts match ?');
497	            params.push(match);
498	        }
499	        // 只在已探测成功的文件里检索：没有时长的行既过不了硬过滤，也不该被选中
500	        where.push("f.probe_state = 'ok'");
501	        // 限定在用户这次选的素材里。**不做这一步，整库的旧素材都会混进来**
502	        if (filters.paths !== undefined) {
503	            if (filters.paths.length === 0) return [];
504	            const clauses = filters.paths.map(
505	                () => "(f.path = ? or f.path like ? escape '\\')"
506	            );
507	            where.push(`(${clauses.join(' or ')})`);
508	            for (const item of filters.paths) {
509	                params.push(item);
510	                // like 的通配符要转义：路径里出现 % 或 _ 并不罕见
511	                params.push(`${escapeLike(item)}/%`);
512	            }
513	        }
514	        if (filters.minDurationMs !== undefined) {
515	            where.push('s.duration_ms >= ?');
516	            params.push(filters.minDurationMs);
517	        }
518	        if (filters.minWidth !== undefined) {
519	            where.push('f.width >= ?');
520	            params.push(filters.minWidth);
521	        }
522	        if (filters.minHeight !== undefined) {
523	            where.push('f.height >= ?');
524	            params.push(filters.minHeight);
525	        }
526	        if (filters.hasAudio !== undefined) {
527	            where.push('f.has_audio = ?');
528	            params.push(filters.hasAudio ? 1 : 0);
529	        }
530	
531	        const selectScore = match !== null ? 'bm25(asset_fts)' : '0';
532	        // **必须带 tie-break**：BM25 打平在小库上很常见（B5 的用例里一抓一大把），
533	        // 而 SQLite 对同分行的顺序不作保证。少了 `s.id`，同一份输入两次调用
534	        // 就可能返回不同的 Top-N，B5「结果稳定」那条 DoD 直接破掉
535	        const orderBy =
536	            match !== null
537	                ? 'score asc, s.id asc'
538	                : 's.duration_ms desc, s.id asc';
539	        const from =
540	            match !== null
541	                ? `asset_fts
542	                   join asset_segments s on s.id = asset_fts.segment_id
543	                   join media_files    f on f.id = s.file_id`
544	                : `asset_segments s
545	                   join media_files f on f.id = s.file_id`;
546	
547	        params.push(limit);
548	        const rows = this.db.all(
549	            `select s.id, s.file_id, s.start_ms, s.duration_ms, s.description, s.tags,
550	                    f.path, f.width, f.height, f.has_audio, ${selectScore} as score
551	               from ${from}
552	              where ${where.join(' and ')}
553	              order by ${orderBy}
554	              limit ?`,
555	            params
556	        ) as Row[];
557	
558	        return rows.map((r) => ({
559	            segmentId: String(r['id']),
560	            fileId: Number(r['file_id']),
561	            path: String(r['path']),
562	            startMs: Number(r['start_ms']),
563	            durationMs: Number(r['duration_ms']),
564	            description: asString(r['description']),
565	            tags: parseTags(r['tags']),
566	            width: asInt(r['width']),
567	            height: asInt(r['height']),
568	            hasAudio: intToBool(r['has_audio']),
569	            score: Number(r['score'])
570	        }));
571	    }

44	    if (typeof value !== 'string') return null;
45	    const parts = value.split('/');
46	    const numerator = asNumber(parts[0]);
